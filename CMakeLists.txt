cmake_minimum_required(VERSION 3.20)
project(HexChess VERSION 0.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(hexchess_rules shared/src/geometry.cpp shared/src/hex_rules.cpp shared/variants/glinski.cpp shared/variants/mccooey.cpp shared/variants/shafran.cpp shared/variants/brusky.cpp shared/variants/devasa.cpp shared/variants/variant.cpp)
target_include_directories(hexchess_rules PUBLIC shared/include)
target_sources(hexchess_rules PRIVATE shared/variants/wellisch.cpp shared/src/wellisch_rules.cpp)
add_executable(engine-one engines/engine-1/main.cpp engines/engine-1/protocol.cpp engines/engine-1/search.cpp)
target_include_directories(engine-one PRIVATE shared/third_party)
target_link_libraries(engine-one PRIVATE hexchess_rules)
set_target_properties(engine-one PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/engines/engine-1"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/engines/engine-1"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/engines/engine-1")
if(MINGW)
  target_link_options(engine-one PRIVATE -static-libgcc -static-libstdc++)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(geometry-tests tests/rules/geometry.cpp)
  target_link_libraries(geometry-tests PRIVATE hexchess_rules)
  add_test(NAME geometry COMMAND geometry-tests)
  add_executable(glinski-tests tests/rules/glinski.cpp engines/engine-1/search.cpp)
  target_link_libraries(glinski-tests PRIVATE hexchess_rules)
  add_test(NAME glinski COMMAND glinski-tests)
  add_executable(mccooey-tests tests/rules/mccooey.cpp engines/engine-1/search.cpp)
  target_link_libraries(mccooey-tests PRIVATE hexchess_rules)
  add_test(NAME mccooey COMMAND mccooey-tests)
endif()

add_executable(engine-two engines/engine-2/main.cpp engines/engine-1/protocol.cpp engines/engine-2/search.cpp)
target_include_directories(engine-two PRIVATE shared/third_party)
target_compile_definitions(engine-two PRIVATE HEXCHESS_ENGINE_IDENTITY="Positional 1.0.0")
target_link_libraries(engine-two PRIVATE hexchess_rules)
set_target_properties(engine-two PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/engines/engine-2")

add_executable(engine-three engines/engine-3/main.cpp engines/engine-1/protocol.cpp engines/engine-3/search.cpp)
target_include_directories(engine-three PRIVATE shared/third_party)
target_compile_definitions(engine-three PRIVATE HEXCHESS_ENGINE_IDENTITY="Defensive 1.0.0" HEXCHESS_SEARCH_CAPABILITY="paranoid")
target_link_libraries(engine-three PRIVATE hexchess_rules)
foreach(pair "engine-two;engine-2" "engine-three;engine-3")
  list(GET pair 0 target)
  list(GET pair 1 folder)
  set_target_properties(${target} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/engines/${folder}"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/engines/${folder}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/engines/${folder}")
  if(MINGW)
    target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
  endif()
endforeach()

if(BUILD_TESTING)
  foreach(pair "positional;engine-2" "defensive;engine-3")
    list(GET pair 0 name)
    list(GET pair 1 folder)
    add_executable(${name}-search-tests tests/engine/search-quality.cpp engines/${folder}/search.cpp)
    target_link_libraries(${name}-search-tests PRIVATE hexchess_rules)
    add_test(NAME ${name}-search COMMAND ${name}-search-tests)
  endforeach()
endif()
